#include <stdio.h>

int reverseNumber(int n) {
    int num = 0;
    while(n > 0){
        num *= 10;
        num += n % 10;
        n /= 10;
    }
    return num;
}

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);

    int result = reverseNumber(n);
    printf("Reversed number: %d\n", result);

    return 0;
}

#include <stdio.h>

int fibonacci(int n) {
    if(n == 0) return 0;
    if(n == 1) return 1;

    return fibonacci(n - 1) + fibonacci(n - 2);
}

int fibonacciSum(int n) {
    int sum = 0;
    for (int i = 0; i <= n; ++i) {
        sum += fibonacci(i);
    }
    return sum;
}

int main() {
    int num;
    printf("Please input a number: ");
    scanf("%d", &num);

    printf("The sum of Fibonacci(0) to Fibonacci(%d) is: %d\n", num, fibonacciSum(num));

    return 0;
}

#include <stdio.h>

int mod_inverse(int a, int b) {
	int m0 = b, x = 1, y = 0;
    while (b != 0) {
        int q = a / b, t = b;
        b = a % b;
        a = t;

        t = y;
        y = x - q * y;
        x = t;
    }
    if(a != 1) return -1;
    return (x + m0) % m0;
}

int main() {
    int a, b;
    printf("Enter the number: ");
    scanf("%d", &a);
    printf("Enter the modulo: ");
    scanf("%d", &b);

    int inv = mod_inverse(a, b);
    if (inv == -1) {
        printf("Inverse not exist.\n");
    } else {
        printf("Result: %d\n", inv);
    }

    return 0;
}

.text
mod_inverse:
    addi    $sp, $sp, -8
    sw      $a0, 0($sp)
    sw      $a1, 4($sp)
    add     $t0, $zero, $a1     # m0 = b
    addi    $t1, $zero, 1       # x = 1
    add     $t2, $zero, $zero   # y = 0
Loop:
    slt     $t3, $zero, $a1     # if b <= 0, jump Exit
    beq     $t3, $zero, Exit    
    add     $t4, $zero, $a1     # t = b
    div     $a0, $a1            
    mflo    $t5                 # q = a / b
    mfhi    $a1                 # b = a % b
    add     $a0, $zero, $t4     # a = t
    add     $t4, $zero, $t2     # t = y
    mul     $t6, $t5, $t2       # q * y
    sub     $t2, $t1, $t6       # y = x - q * y
    add     $t1, $zero, $t4     # x = t
    j Loop
Exit:
    addi    $t7, $zero, 1
    bne     $a0, $t7, NotExist  # if (a != 1) return -1
    add     $t1, $t1, $t0       # x = x + m0
    div     $t1, $t0            
    mfhi    $v0                 # x % m0
    lw      $a0, 0($sp)
    lw      $a1, 4($sp)
    addi    $sp, $sp, 8
	jr 		$ra					# return to caller
NotExist:
    addi    $v0, $zero, -1
    lw      $a0, 0($sp)
    lw      $a1, 4($sp)
    addi    $sp, $sp, 8
	jr 		$ra					# return to caller
